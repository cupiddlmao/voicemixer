cmake_minimum_required(VERSION 3.20)
project(LinuxMixer VERSION 0.4.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=.")
endif()
find_package(Qt6 QUIET COMPONENTS Core Gui Widgets Network)
add_executable(linux-mixer tests/verify_profile.cpp native/profile_bundle.cpp native/profile_ui.cpp native/main.cpp native/recorder.cpp native/recorder_ui.cpp native/recorder_test.cpp native/paths.cpp native/hardware_links.cpp native/engine.cpp native/console.cpp native/smoke.cpp native/routing.cpp native/routing_settings.cpp native/app_routing_test.cpp native/plugin_racks.cpp native/audio_health.cpp native/health_ui.cpp native/update_checker.cpp native/release_info.cpp)
target_include_directories(linux-mixer PRIVATE native)
target_compile_definitions(linux-mixer PRIVATE VOICEMIXER_VERSION="${PROJECT_VERSION}")
target_compile_options(linux-mixer PRIVATE -Wall -Wextra -Wpedantic)
add_library(mixer_dsp MODULE native/dsp_ladspa.cpp)
set_target_properties(mixer_dsp PROPERTIES PREFIX "lib" OUTPUT_NAME "mixer_dsp")
target_compile_options(mixer_dsp PRIVATE -Wall -Wextra -Wpedantic)
if(Qt6_FOUND)
  target_link_libraries(linux-mixer PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
else()
  # Optional local SDK extracted from official distro development packages.
  set(LOCAL_QT "${CMAKE_CURRENT_SOURCE_DIR}/.deps/sdk/usr/include/x86_64-linux-gnu/qt6")
  if(NOT EXISTS "${LOCAL_QT}/QtWidgets/QApplication")
    message(FATAL_ERROR "Install qt6-base-dev (Ubuntu) or qt6-base (Arch), or prepare .deps/sdk.")
  endif()
  target_include_directories(linux-mixer PRIVATE "${LOCAL_QT}" "${LOCAL_QT}/QtCore" "${LOCAL_QT}/QtGui" "${LOCAL_QT}/QtWidgets" "${LOCAL_QT}/QtNetwork")
  foreach(component Core Gui Widgets Network)
    find_library(QT_${component} NAMES Qt6${component} libQt6${component}.so.6 REQUIRED)
    target_link_libraries(linux-mixer PRIVATE ${QT_${component}})
  endforeach()
endif()

include(CTest)
if(BUILD_TESTING)
  add_test(NAME complete-profiles COMMAND linux-mixer --profile-test)
  set_tests_properties(complete-profiles PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  add_executable(verify-dsp tests/verify_dsp.cpp)
  target_compile_options(verify-dsp PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
  add_test(NAME dsp COMMAND verify-dsp)
  if(Qt6_FOUND)
    add_executable(verify-stability tests/verify_stability.cpp native/audio_health.cpp native/update_checker.cpp native/release_info.cpp native/paths.cpp)
    target_include_directories(verify-stability PRIVATE native)
    target_link_libraries(verify-stability PRIVATE Qt6::Core Qt6::Widgets Qt6::Network)
    target_compile_options(verify-stability PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
    add_test(NAME stability-and-updates COMMAND verify-stability)
  endif()
endif()

option(BUILD_RNNOISE "Build the vendored RNNoise LADSPA plugin" ON)
if(BUILD_RNNOISE)
  enable_language(C)
  # The upstream build's CMake minimum predates CMake 4's policy compatibility.
  set(CMAKE_POLICY_VERSION_MINIMUM 3.6 CACHE STRING "" FORCE)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  foreach(format VST VST3 LV2 AU AUV3)
    set(BUILD_${format}_PLUGIN OFF CACHE BOOL "" FORCE)
  endforeach()
  set(BUILD_LADSPA_PLUGIN ON CACHE BOOL "" FORCE)
  add_subdirectory(vendor/rnnoise-plugin EXCLUDE_FROM_ALL)
  set_target_properties(rnnoise_ladspa PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  add_dependencies(linux-mixer rnnoise_ladspa mixer_dsp)
endif()

include(GNUInstallDirs)
install(TARGETS linux-mixer RUNTIME DESTINATION bin)
install(PROGRAMS packaging/assets/voicemixer DESTINATION bin)
install(TARGETS mixer_dsp LIBRARY DESTINATION lib/linux-mixer)
if(BUILD_RNNOISE)
  install(TARGETS rnnoise_ladspa LIBRARY DESTINATION lib/linux-mixer)
endif()
install(FILES packaging/assets/linux-mixer.desktop DESTINATION share/applications)
install(FILES packaging/assets/linux-mixer.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES docs/PROFILES.md docs/RECORDER.md docs/PLUGINS.md docs/STABILITY.md docs/UPDATING.md DESTINATION share/doc/linux-mixer/docs)
install(FILES README.md LICENSE DESTINATION share/doc/linux-mixer)
install(FILES vendor/README.md DESTINATION share/doc/linux-mixer RENAME THIRD-PARTY.md)
install(FILES vendor/rnnoise-plugin/LICENSE DESTINATION share/doc/linux-mixer RENAME RNNoise-LICENSE)
set(CPACK_GENERATOR "DEB")
set(CPACK_PACKAGE_NAME "linux-mixer")
if(EXISTS "/etc/os-release")
  file(STRINGS "/etc/os-release" OS_RELEASE_VERSION REGEX "^VERSION_ID=")
  string(REGEX REPLACE "VERSION_ID=\"?([^\"]+)\"?" "\\1" HOST_DISTRO_VERSION "${OS_RELEASE_VERSION}")
else()
  set(HOST_DISTRO_VERSION "native")
endif()
set(CPACK_PACKAGE_FILE_NAME "linux-mixer-0.4.1-ubuntu${HOST_DISTRO_VERSION}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_PACKAGE_VERSION "0.4.1")
set(CPACK_PACKAGE_CONTACT "Linux Mixer contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Personal PipeWire audio mixing console")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "pipewire (>= 1.0), pipewire-bin (>= 1.0), pipewire-pulse, wireplumber, pulseaudio-utils, ffmpeg, zam-plugins, qt6-wayland")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
include(CPack)

option(BUILD_PLUGIN_HOST "Build the isolated native VST3/LADSPA input rack" ON)
if(BUILD_PLUGIN_HOST)
  add_subdirectory(native/plugins)
endif()

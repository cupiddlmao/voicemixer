find_package(PkgConfig REQUIRED)
pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
set(JUCE_MODULES_ONLY ON CACHE BOOL "" FORCE)
add_subdirectory(${PROJECT_SOURCE_DIR}/vendor/rnnoise-plugin/external/JUCE ${CMAKE_CURRENT_BINARY_DIR}/juce EXCLUDE_FROM_ALL)
add_library(mixer_plugin_runtime STATIC)
set_target_properties(mixer_plugin_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_executable(linux-mixer-plugin-host host.cpp)
target_link_libraries(linux-mixer-plugin-host PRIVATE mixer_plugin_runtime)
target_include_directories(mixer_plugin_runtime PUBLIC ${PROJECT_SOURCE_DIR}/vendor/rnnoise-plugin/src/ladspa_plugin)
target_compile_definitions(mixer_plugin_runtime PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_STANDALONE_APPLICATION=1 JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_LADSPA=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
target_compile_options(mixer_plugin_runtime PUBLIC -include utility)
target_link_libraries(mixer_plugin_runtime PRIVATE juce::juce_audio_processors juce::juce_gui_extra PUBLIC PkgConfig::PIPEWIRE)
target_include_directories(mixer_plugin_runtime PUBLIC ${PROJECT_SOURCE_DIR}/vendor/rnnoise-plugin/external/JUCE/modules ${PROJECT_SOURCE_DIR}/vendor/rnnoise-plugin/external/JUCE/modules/juce_audio_processors/format_types/VST3_SDK)
set_target_properties(linux-mixer-plugin-host PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
install(TARGETS linux-mixer-plugin-host RUNTIME DESTINATION bin)
